if(NOT ASR_CLIPS AND NOT CMAKE_BUILD_EARLY_EXPANSION)
    message(FATAL_ERROR "pass -D ASR_CLIPS=<folder with NN.audio from tools/golden2raw.py>")
endif()
file(GLOB clips ${ASR_CLIPS}/*.audio)
list(SORT clips)
set(i 0)
set(embed "")
foreach(c ${clips})
    configure_file(${c} ${CMAKE_CURRENT_BINARY_DIR}/clip${i}.bin COPYONLY)
    list(APPEND embed ${CMAKE_CURRENT_BINARY_DIR}/clip${i}.bin)
    math(EXPR i "${i} + 1")
endforeach()
idf_component_register(SRCS bench.c PRIV_REQUIRES asr vform esp_timer esp_partition spi_flash
                       EMBED_FILES ${embed})
target_compile_definitions(${COMPONENT_LIB} PRIVATE N_CLIPS=${i})
